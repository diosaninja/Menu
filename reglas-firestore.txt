rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /cocinas/{cocina} {
      allow get, create, update: if request.auth != null && cocina.size() == 16;
      allow list, delete: if false;
      match /{coleccion}/{doc} {
        allow read, write: if request.auth != null && cocina.size() == 16;
      }
    }
  }
}
